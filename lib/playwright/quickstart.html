Quickstart & Recipes

Non-root Chromium automation: screenshots, scraping, PDF, and form interaction on Android

Recipe 1: Screenshot Capture (Python)

import asyncio
from termux_playwright import async_playwright

async def capture_screenshot():
    async with async_playwright() as p:
        browser = await p.chromium.launch(
            headless=True,
            args=['--no-sandbox', '--disable-gpu']
        )
        page = await browser.new_page()
        await page.goto('https://example.com', wait_until='networkidle')

        # Save to Samsung Gallery-indexed path
        await page.screenshot(
            path='/data/data/com.termux/files/home/storage/pictures/screenshot.png',
            full_page=True
        )
        await browser.close()
        print("Screenshot saved to Gallery")

asyncio.run(capture_screenshot())

Recipe 2: Web Scraping with Anti-Bot Evasion

from termux_playwright import sync_playwright
from termux_playwright.stealth import stealth_sync

with sync_playwright() as p:
    browser = p.chromium.launch(headless=True, args=['--no-sandbox'])
    page = browser.new_page()

    # Apply stealth patches (removes webdriver fingerprint)
    stealth_sync(page)

    page.goto('https://target-site.com')
    page.wait_for_selector('.product-list')

    items = page.query_selector_all('.product-item')
    for item in items:
        print({
            'name': item.query_selector('.name').inner_text(),
            'price': item.query_selector('.price').inner_text()
        })

    browser.close()

Recipe 3: PDF Generation

from termux_playwright import sync_playwright

with sync_playwright() as p:
    browser = p.chromium.launch(headless=True, args=['--no-sandbox'])
    page = browser.new_page()
    page.goto('https://docs.example.com/report')
    page.wait_for_load_state('networkidle')

    page.pdf(
        path='/data/data/com.termux/files/home/report.pdf',
        format='A4',
        print_background=True,
        margin={'top': '1cm', 'bottom': '1cm', 'left': '1cm', 'right': '1cm'}
    )
    browser.close()
    print("PDF saved")

Recipe 4: Node.js / TypeScript

import { chromium, Page } from 'termux-playwright';

async function automateForm(): Promise<void> {
  const browser = await chromium.launch({
    headless: true,
    args: ['--no-sandbox', '--disable-gpu']
  });

  const page: Page = await browser.newPage();
  await page.goto('https://form.example.com');

  await page.fill('#username', 'mobile_agent');
  await page.fill('#password', 'secure_pass');
  await page.click('[type="submit"]');

  await page.waitForURL('**/dashboard');
  const title = await page.title();
  console.log('Logged in. Page title:', title);

  await browser.close();
}

automateForm().catch(console.error);

Recipe 5: Batch Parallel Crawling (ARM64 Optimized)

import asyncio
from termux_playwright import async_playwright

URLS = [
    'https://site1.example.com',
    'https://site2.example.com',
    'https://site3.example.com',
]

async def crawl_url(page, url: str) -> dict:
    await page.goto(url, timeout=15000)
    return {'url': url, 'title': await page.title()}

async def parallel_crawl():
    async with async_playwright() as p:
        browser = await p.chromium.launch(
            headless=True,
            args=['--no-sandbox']
        )
        # Allocate parallel pages (keep ≤ 3 on 4GB RAM devices)
        pages = [await browser.new_page() for _ in range(min(3, len(URLS)))]
        tasks = [crawl_url(pages[i % len(pages)], url) for i, url in enumerate(URLS)]
        results = await asyncio.gather(*tasks)
        await browser.close()
        return results

results = asyncio.run(parallel_crawl())
for r in results:
    print(r)
