API Reference

Complete Python and Node.js API for non-root ARM64 Chromium automation on Termux

Module: termux_playwright

Context Manager: async_playwright()

from termux_playwright import async_playwright

async with async_playwright() as p:
    browser = await p.chromium.launch(
        headless=True,
        args=['--no-sandbox', '--disable-gpu'],
        timeout=30000
    )
    # browser: BrowserType

Browser: p.chromium.launch() Parameters

ParameterTypeDefaultDescription
headlessboolTrueRun Chromium headless (required on Termux)
argslist[str][]Chromium CLI flags (always include --no-sandbox)
executable_pathstrautoOverride Chromium binary path
timeoutint30000Browser launch timeout (ms)
proxydictNone{'server': '...', 'username': '...', 'password': '...'}
ignore_default_argslist[]Remove specific default Playwright args

Page: browser.new_page() / Core Methods

page = await browser.new_page()

# Navigation
await page.goto(url, wait_until='networkidle', timeout=15000)
# wait_until: 'load' | 'domcontentloaded' | 'networkidle' | 'commit'

# Selectors
element = await page.query_selector('.css-selector')
elements = await page.query_selector_all('div.item')
await page.wait_for_selector('#login-form', timeout=5000)

# Interaction
await page.fill('#input-field', 'text value')
await page.click('[type="submit"]')
await page.select_option('#dropdown', 'value')
await page.keyboard.press('Enter')

# Evaluation
result = await page.evaluate('() => document.title')
text = await page.inner_text('.content')

# Screenshot / PDF
await page.screenshot(path='shot.png', full_page=True, type='png')
await page.pdf(path='doc.pdf', format='A4')

# Network
await page.route('**/*.{png,jpg}', lambda r: r.abort())
await page.route('**/api/**', lambda r: r.fulfill(body='{"mock":true}'))

Browser Context: Isolation & Fingerprinting

context = await browser.new_context(
    viewport={'width': 1080, 'height': 2340},
    user_agent='Mozilla/5.0 (Linux; Android 13; SM-S901B)',
    locale='ko-KR',
    timezone_id='Asia/Seoul',
    geolocation={'latitude': 37.5665, 'longitude': 126.9780},
    permissions=['geolocation'],
    extra_http_headers={'Accept-Language': 'ko-KR'}
)
page = await context.new_page()

Node.js / TypeScript API

import { chromium, Browser, Page, BrowserContext } from 'termux-playwright';

const browser: Browser = await chromium.launch({ headless: true, args: ['--no-sandbox'] });
const page: Page = await browser.newPage();
await page.goto('https://example.com');
const title: string = await page.title();
await browser.close();
