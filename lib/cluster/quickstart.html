Basic Execution Recipe

from ameva_cluster import AmevaCluster, ClusterMaster

# 1. Start worker daemon with Guard protection on remote phone:
# ameva-cluster worker --port 50052

# 2. Execute distributed inference across phone fleet:
AmevaCluster.run_master(
    model_path="Qwen2.5-7B-Instruct-Q4_K_M.gguf",
    workers=["100.106.251.21:50052", "100.77.47.37:50052"],
    tensor_split="40,30,30",
    prompt="Explain disaggregated neural memory architecture."
)
