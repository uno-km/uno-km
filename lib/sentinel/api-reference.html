API Reference

Complete TypeScript/JavaScript API — Sentinel class, configuration, evaluation result, and utilities

Class: Sentinel

import { Sentinel } from 'ameva-sentinel';

const sentinel = new Sentinel(config: SentinelConfig);

Constructor Parameters

ParameterTypeRequiredDefaultDescription
hmacSecretstring✅—256-bit entropy HMAC signing secret
riskThresholdnumber❌65Block if score ≥ threshold (0–100)
shadowModeboolean❌falseLog-only mode, never blocks
tokenTTLnumber❌300HMAC token validity (seconds)
weightsSignalWeights❌balancedPer-axis risk signal weights (sum=1.0)
edgeProviderstring❌'auto'CDN provider for IP extraction
trustedProxiesstring[]❌[]CIDR ranges for trusted reverse proxies

Method: sentinel.evaluate(request)

const result: EvaluationResult = await sentinel.evaluate(req);
// req: IncomingMessage (Node.js) | Request (Web API / Edge)

Returns: EvaluationResult

FieldTypeDescription
scorenumber (0–100)Composite risk score. Higher = riskier.
tokenstringHMAC-SHA256 signed evaluation token (include in response headers)
blockedbooleanTrue if score ≥ threshold AND not in shadow mode
signalsSignalBreakdownPer-axis score breakdown for observability
latencyMsnumberInternal evaluation latency in milliseconds
requestIdstringUUID for correlation with access logs

Method: sentinel.middleware()

// Express-compatible middleware factory
app.use(sentinel.middleware());

// Custom handler for blocked requests
app.use(sentinel.middleware({
  onBlocked: (req, res, result) => {
    res.status(429).json({ error: 'Blocked', score: result.score });
  },
  onEvaluation: (req, result) => {
    console.log('Eval:', result.score, result.requestId);
  }
}));

Utility: verifyToken(token, secret, options?)

import { verifyToken } from 'ameva-sentinel';

const isValid: boolean = verifyToken(
  token: string,
  secret: string,
  options?: {
    maxAge?: number;    // Max token age in seconds (default: tokenTTL)
    minScore?: number;  // Minimum acceptable score (default: 0)
    maxScore?: number;  // Maximum acceptable score (default: 100)
  }
);

Interface: SignalBreakdown

interface SignalBreakdown {
  headerEntropy: number;    // 0–100: HTTP header structure entropy analysis
  tlsFingerprint: number;   // 0–100: JA3/JA3S fingerprint risk
  ipReputation: number;     // 0–100: IP reputation and ASN classification
  behaviorPattern: number;  // 0–100: Request timing and sequencing pattern
  geoConsistency: number;   // 0–100: IP geolocation vs Accept-Language mismatch
  composite: number;        // Final weighted composite score
}
