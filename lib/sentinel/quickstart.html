Quickstart & Integration Recipes

5-minute integration patterns for Express, Next.js, FastAPI, and CDN deployments

Shadow Mode Recommended for Initial Deployment

Set shadowMode: true on first deploy to observe risk scores without blocking requests. Switch to enforcement mode after validating your threshold calibration.

Recipe 1: Express.js Middleware (Node.js)

Drop-in middleware that evaluates every incoming request and attaches the risk score to req.sentinel:

import express from 'express';
import { Sentinel, createExpressMiddleware } from 'ameva-sentinel';

const app = express();

// 1. Initialize Sentinel with HMAC secret
const sentinel = new Sentinel({
  hmacSecret: process.env.SENTINEL_SECRET,
  riskThreshold: 65,     // Block requests with score >= 65
  shadowMode: false,     // true = observe only, false = enforce
  tokenTTL: 300          // HMAC token validity in seconds
});

// 2. Register as Express middleware
app.use(createExpressMiddleware(sentinel));

// 3. Access evaluation result in route handlers
app.post('/api/submit', (req, res) => {
  const { score, token, signals } = req.sentinel;

  if (score >= 65) {
    return res.status(429).json({ error: 'High-risk request blocked', score });
  }

  res.json({ status: 'accepted', score, token });
});

app.listen(3000);

Recipe 2: Next.js Edge Middleware (Vercel)

Deploy at the Vercel Edge with zero cold-start latency:

// middleware.ts (Next.js project root)
import { NextRequest, NextResponse } from 'next/server';
import { Sentinel } from 'ameva-sentinel';

const sentinel = new Sentinel({
  hmacSecret: process.env.SENTINEL_SECRET!,
  riskThreshold: 70,
  shadowMode: false
});

export async function middleware(request: NextRequest) {
  const result = await sentinel.evaluate(request);

  if (result.score >= 70) {
    return NextResponse.json(
      { error: 'Request blocked by Sentinel', score: result.score },
      { status: 429 }
    );
  }

  const response = NextResponse.next();
  response.headers.set('X-Sentinel-Score', result.score.toString());
  response.headers.set('X-Sentinel-Token', result.token);
  return response;
}

export const config = {
  matcher: ['/api/:path*', '/submit/:path*']
};

Recipe 3: Browser CDN — Client-Side Evaluation

<script src="https://cdn.jsdelivr.net/npm/ameva-sentinel/dist/sentinel.min.js"></script>
<script>
  const sentinel = new Sentinel.BrowserClient({
    endpoint: 'https://your-api.example.com/api/sentinel',
    autoCollect: true   // Automatically evaluates on DOM ready
  });

  document.querySelector('#submit-btn').addEventListener('click', async () => {
    const { token, score } = await sentinel.getToken();

    // Include token in your form submission
    const response = await fetch('/api/submit', {
      method: 'POST',
      headers: {
        'X-Sentinel-Token': token,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ data: 'payload' })
    });
  });
</script>

Recipe 4: FastAPI Python Integration

from fastapi import FastAPI, Request, HTTPException
from ameva_sentinel import Sentinel, SentinelConfig

app = FastAPI()

sentinel = Sentinel(SentinelConfig(
    hmac_secret="your-secret-256bit",
    risk_threshold=65,
    shadow_mode=False
))

@app.middleware("http")
async def sentinel_middleware(request: Request, call_next):
    result = await sentinel.evaluate(request)

    if result.score >= 65 and not sentinel.config.shadow_mode:
        raise HTTPException(status_code=429, detail={
            "error": "High-risk request blocked",
            "score": result.score
        })

    request.state.sentinel = result
    response = await call_next(request)
    response.headers["X-Sentinel-Score"] = str(result.score)
    return response

Recipe 5: Standalone Token Verification

Verify HMAC tokens generated by the client-side SDK on your server:

import { verifyToken } from 'ameva-sentinel';

// Verify token from incoming request header
const token = req.headers['x-sentinel-token'];
const isValid = verifyToken(token, process.env.SENTINEL_SECRET, {
  maxAge: 300,          // Reject tokens older than 5 minutes
  minScore: 0,          // Accept all risk levels (enforce separately)
  maxScore: 100
});

if (!isValid) {
  return res.status(401).json({ error: 'Invalid or expired Sentinel token' });
}

Recipe 6: 1-Line Embedded Observability Dashboard (/sentinel/dashboard)

Serve the full real-time forensic observability & risk dashboard directly inside your application with zero external hosting dependencies:

// 1. Express (TypeScript / Node.js)
import express from 'express';
import { createExpressDashboardHandler } from '@ameva/sentinel/dashboard';

const app = express();
app.use('/sentinel/dashboard', createExpressDashboardHandler());

// 2. Fastify (TypeScript / Node.js)
import Fastify from 'fastify';
import { createFastifyDashboardHandler } from '@ameva/sentinel/dashboard';

const fastify = Fastify();
fastify.get('/sentinel/dashboard', createFastifyDashboardHandler());

// 3. Next.js App Router (app/sentinel/dashboard/route.ts)
import { createNextDashboardHandler } from '@ameva/sentinel/dashboard';
export const GET = createNextDashboardHandler();

// 4. FastAPI (Python)
from fastapi import FastAPI
from ameva_sentinel import mount_fastapi_dashboard

app = FastAPI()
mount_fastapi_dashboard(app, path="/sentinel/dashboard")

// 5. Flask (Python)
from flask import Flask
from ameva_sentinel import mount_flask_dashboard

app = Flask(__name__)
mount_flask_dashboard(app, path="/sentinel/dashboard")
