Advanced Parameters & Trust Boundary Configuration

Shadow mode, multi-CDN edge routing, risk axis tuning, and HMAC secret management

1. Full Configuration Schema

interface SentinelConfig {
  // Required
  hmacSecret: string;         // 256-bit entropy minimum (use crypto.randomBytes(32))

  // Risk Control
  riskThreshold: number;      // 0-100. Block if score >= threshold. Default: 65
  shadowMode: boolean;        // true = log only, false = enforce. Default: false
  tokenTTL: number;           // HMAC token validity seconds. Default: 300

  // Signal Axis Weights (sum must equal 1.0)
  weights?: {
    headerEntropy: number;    // HTTP header structure entropy. Default: 0.25
    tlsFingerprint: number;   // JA3/JA3S fingerprint score. Default: 0.20
    ipReputation: number;     // IP reputation & ASN analysis. Default: 0.25
    behaviorPattern: number;  // Request timing/sequencing. Default: 0.20
    geoConsistency: number;   // IP geo vs Accept-Language. Default: 0.10
  };

  // Edge Provider Settings
  edgeProvider?: 'vercel' | 'cloudflare' | 'fastly' | 'akamai' | 'auto';
  trustedProxies?: string[];  // CIDR ranges for trusted reverse proxies

  // Observability
  onEvaluation?: (result: EvaluationResult) => void;  // Callback for every evaluation
  logLevel?: 'none' | 'error' | 'warn' | 'debug';
}

2. Shadow Mode — Safe Production Rollout

// Phase 1: Shadow mode — observe without blocking
const sentinel = new Sentinel({
  hmacSecret: process.env.SENTINEL_SECRET,
  shadowMode: true,         // Never blocks requests
  riskThreshold: 65,
  onEvaluation: (result) => {
    // Log to observability platform
    analytics.track('sentinel.evaluation', {
      score: result.score,
      signals: result.signals,
      blocked: result.score >= 65  // Would have been blocked
    });
  }
});

// Phase 2: After 2-week baseline collection, switch to enforcement
// sentinel.shadowMode = false;

3. Custom Signal Axis Weights

// High-security API: Weight IP reputation heavily
const highSecuritySentinel = new Sentinel({
  hmacSecret: process.env.SENTINEL_SECRET,
  riskThreshold: 55,
  weights: {
    headerEntropy: 0.20,
    tlsFingerprint: 0.25,
    ipReputation: 0.35,    // Elevated for high-value endpoints
    behaviorPattern: 0.15,
    geoConsistency: 0.05
  }
});

// Low-friction consumer API: Weight behavioral pattern lightly
const consumerSentinel = new Sentinel({
  hmacSecret: process.env.SENTINEL_SECRET,
  riskThreshold: 75,        // More permissive threshold
  weights: {
    headerEntropy: 0.30,
    tlsFingerprint: 0.15,
    ipReputation: 0.25,
    behaviorPattern: 0.20,
    geoConsistency: 0.10
  }
});

4. Trusted Proxy Configuration (Multi-CDN)

const sentinel = new Sentinel({
  hmacSecret: process.env.SENTINEL_SECRET,
  edgeProvider: 'auto',        // Auto-detect CDN from headers
  trustedProxies: [
    '173.245.48.0/20',         // Cloudflare
    '103.21.244.0/22',
    '76.76.21.0/24',           // Vercel
    '64.18.0.0/20',            // Fastly
  ]
});

5. HMAC Secret Rotation

import { Sentinel, rotateSecret } from 'ameva-sentinel';

// Graceful secret rotation: accepts both old and new secret during TTL window
const sentinel = new Sentinel({
  hmacSecret: process.env.SENTINEL_SECRET_NEW,
  legacySecrets: [process.env.SENTINEL_SECRET_OLD],  // Accepted until TTL expires
  tokenTTL: 300
});

// Generate new secret
const newSecret = await rotateSecret(32);  // 32 bytes = 256 bits
console.log('New HMAC secret:', newSecret.toString('hex'));
