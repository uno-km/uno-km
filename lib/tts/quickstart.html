Command-Line Interface (CLI) Recipes

# 1. High-Resolution Studio Neural Synthesis with Speaker Playback
termux-tts synth -e vulkan --tier high -t "Operating at full hardware capacity." -o speech.wav --play

# 2. Instant Zero-Dependency DSP Formant Synthesis
termux-tts synth -e dsp -p ultra -t "Zero dependency parametric speech synthesis." -o dsp.wav

# 3. Direct Android Hardware Speaker Broadcast
termux-tts speak -t "System notification broadcast." -l en

# 4. Hardware and Environment Diagnostics
termux-tts doctor

Python SDK Canon

import termux_tts as tts

# Initialize pure Vulkan GPU neural synthesis engine
with tts.load(engine="vulkan", model_tier="high") as engine:
    res = engine.synthesize("Validating deterministic tensor execution.", output="out.wav")
    print(f"Elapsed: {res.elapsed_ms:.1f}ms | RTF: {res.rtf:.4f}x")
