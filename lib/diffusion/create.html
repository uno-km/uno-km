import termux_diffusion as td

# 1. Initialize quantized hardware engine
engine = td.DiffusionEngine(model="sd-turbo", threads=8)

# 2. Execute on-device inference with Samsung Gallery auto-indexing
result = engine.generate(
    prompt="cute orange cat surfing on a small wave, centered composition, cat clearly visible in the middle of the image, wet reflective ocean surface, playful expression, fluffy fur slightly wet, bright blue sky above, soft white clouds, sunlight reflections on water, dynamic splash effects, cinematic lighting, highly detailed, vibrant colors, adorable, clean composition, focus on the cat, mobile wallpaper, masterpiece, no text, no watermark",
    steps=4,
    cfg_scale=1.5,
    seed=42891,
    width=512,
    height=512,
    save_to_gallery=True
)

print(f"Generated successfully: {result.path} (Time: {result.latency_ms}ms)")
