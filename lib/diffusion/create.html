선택한 프롬프트 및 모델 파라미터가 AMEVA-Forge WebGPU 및 Termux SDK 코드로 실시간 동기화됩니다.

import { ForgeDiffusion } from "@uno-km/ameva-forge";

// 1. Initialize WebGPU Denoising Pipeline with animagine-turbo CDN
const pipeline = await ForgeDiffusion.fromPretrained("cagliostrolab/animagine-xl-3.1", {
  device: "webgpu",
  precision: "fp16"
});

// 2. Execute zero-server 4-step diffusion generation
const image = await pipeline.generate({
  prompt: "cute orange cat surfing on a small wave, centered composition, cat clearly visible in the middle of the image, wet reflective ocean surface, playful expression, fluffy fur slightly wet, bright blue sky above, soft white clouds, sunlight reflections on water, dynamic splash effects, cinematic lighting, highly detailed, vibrant colors, adorable, clean composition, focus on the cat, mobile wallpaper, masterpiece, no text, no watermark",
  steps: 4,
  cfg: 1.5,
  seed: 42891,
  width: 512,
  height: 512
});

image.renderToCanvas(document.getElementById("outputCanvas"));
