Basic Execution Recipe

import ameva_runtime as ameva
from ameva_runtime import vulkan

# 1. Inspect on-device silicon topology
profile = ameva.detect_hardware()
print(f"Detected SoC: {profile.soc_name} | GPU: {profile.gpu_vendor} ({profile.gpu_model})")

# 2. Hardware diagnostic inspection via Vulkan engine
doc = vulkan.Doctor()
report = doc.run_self_test(verbose=False)
print(f"GPU Target: {report.device_name} (Passed: {report.passed_stages}/{report.total_stages})")

# 3. Plan deterministic execution graph
plan = ameva.plan(modality="tts", tier="high")
print(f"Optimal Backend: {plan.recommended_backend} | Subgroup Size: {plan.subgroup_size}")
